function formatTime(seconds) {
  const totalMinutes = Math.floor(seconds / 60);
  const hours = Math.floor(totalMinutes / 60);
  const minutes = totalMinutes % 60;
  const secs = Math.floor(seconds % 60);
  return `${String(hours).padStart(2, '0')}:${String(minutes).padStart(2, '0')}:${String(secs).padStart(2, '0')}`;
}

function render() {
  context.clearRect(0, 0, VIEW_WIDTH, VIEW_HEIGHT);
  if (!world || !running) {
    context.fillStyle = '#07090a';
    context.fillRect(0, 0, VIEW_WIDTH, VIEW_HEIGHT);
    return;
  }
  drawWorld();
  drawHud();
  if (commandMapOpen) drawCommandMap();
  if (diagnosticsOpen) drawDiagnostics();
  if (evolutionOpen) drawEvolution();
  if (helpOpen) drawHelp();
  if (paused && !helpOpen && !evolutionOpen && world.status === 'playing') drawPause();
}

function frame(now) {
  const dt = Math.min(0.05, Math.max(0, (now - lastFrame) / 1000));
  lastFrame = now;
  update(dt);
  render();
  requestAnimationFrame(frame);
}

function handleInteraction() {
  if (!world || paused || commandMapOpen || diagnosticsOpen || helpOpen || evolutionOpen) return;
  const result = interact(world);
  if (result.kind === 'choose_evolution') evolutionOpen = true;
  else if (result.kind === 'need_scrap') addClientEvent(`The Heartforge needs ${result.amount} more Scrap.`, 'warning');
  else if (result.kind === 'none') addClientEvent('Nothing here requires direct intervention.');
}

function cycleCamera() {
  const expeditionExists = activeExpeditionRobots().length > 0;
  if (cameraMode === 'player') cameraMode = 'heartforge';
  else if (cameraMode === 'heartforge' && expeditionExists) cameraMode = 'expedition';
  else cameraMode = 'player';
}

window.addEventListener('keydown', (event) => {
  if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', 'Space', 'Tab'].includes(event.code)) event.preventDefault();
  if (!running || !world) return;

  if (evolutionOpen && ['Digit1', 'Digit2', 'Digit3'].includes(event.code)) {
    const choices = { Digit1: 'mechromancer', Digit2: 'machines', Digit3: 'heartforge' };
    if (chooseEvolution(world, choices[event.code])) evolutionOpen = false;
    return;
  }

  if (event.repeat && !event.code.startsWith('Arrow') && !event.code.startsWith('KeyW') && !event.code.startsWith('KeyA') && !event.code.startsWith('KeyS') && !event.code.startsWith('KeyD')) return;
  keys.add(event.code);

  if (event.code === 'ShiftLeft' || event.code === 'ShiftRight' || event.code === 'Space') dashQueued = true;
  if (event.code === 'KeyE') handleInteraction();
  if (event.code === 'KeyX' && !paused && !evolutionOpen) {
    if (!authorizeExpedition(world)) {
      const missing = Math.max(0, EXPEDITION_COST - world.scrap);
      addClientEvent(missing > 0 ? `The expedition needs ${missing} more Scrap.` : 'The expedition is not available yet.', 'warning');
    }
  }
  if (event.code === 'KeyR' && !paused) emergencyRecall(world);
  if (event.code === 'KeyF') cycleCamera();
  if (event.code === 'KeyM') {
    commandMapOpen = !commandMapOpen;
    diagnosticsOpen = false;
    helpOpen = false;
  }
  if (event.code === 'Tab') {
    diagnosticsOpen = !diagnosticsOpen;
    commandMapOpen = false;
    helpOpen = false;
  }
  if (event.code === 'KeyH') {
    helpOpen = !helpOpen;
    commandMapOpen = false;
    diagnosticsOpen = false;
  }
  if (event.code === 'F5') {
    event.preventDefault();
    saveWorld(true);
  }
  if (event.code === 'F9') {
    event.preventDefault();
    loadWorld();
  }
  if (event.code === 'Escape') {
    if (commandMapOpen || diagnosticsOpen || helpOpen) {
      commandMapOpen = false;
      diagnosticsOpen = false;
      helpOpen = false;
    } else if (!evolutionOpen) {
      paused = !paused;
    }
  }
});

window.addEventListener('keyup', (event) => keys.delete(event.code));
canvas.addEventListener('mousemove', (event) => {
  const rect = canvas.getBoundingClientRect();
  mouse.x = (event.clientX - rect.left) / rect.width * VIEW_WIDTH;
  mouse.y = (event.clientY - rect.top) / rect.height * VIEW_HEIGHT;
});
canvas.addEventListener('contextmenu', (event) => event.preventDefault());
window.addEventListener('blur', () => {
  keys.clear();
  if (running && world?.status === 'playing' && !evolutionOpen) paused = true;
});
window.addEventListener('beforeunload', () => saveWorld(false));

autoBoot();

  loading.hidden = true;
  gameShell.hidden = true;
  endScreen.hidden = true;
  titleScreen.hidden = false;
  continueButton.disabled = !hasSave();
  newWorldButton.addEventListener('click', startNewWorld);
  continueButton.addEventListener('click', loadWorld);
  restartButton.addEventListener('click', startNewWorld);
  loadButton.addEventListener('click', loadWorld);
  document.querySelector('#return-title').addEventListener('click', returnToTitle);
  requestAnimationFrame(frame);
}
