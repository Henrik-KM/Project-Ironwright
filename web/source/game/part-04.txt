function drawLighting() {
  lightContext.clearRect(0, 0, VIEW_WIDTH, VIEW_HEIGHT);
  lightContext.fillStyle = 'rgba(0, 0, 0, 0.68)';
  lightContext.fillRect(0, 0, VIEW_WIDTH, VIEW_HEIGHT);
  lightContext.globalCompositeOperation = 'destination-out';

  const lights = [
    { entity: world.player, radius: 235 },
    { entity: world.heartforge, radius: world.heartforge.lightRadius + 100 },
    ...world.robots.filter((robot) => !robot.disabled).map((entity) => ({ entity, radius: entity.expedition ? 105 : 72 })),
  ];
  for (const light of lights) {
    const point = worldToScreen(light.entity.x, light.entity.y);
    if (point.x < -light.radius || point.y < -light.radius || point.x > VIEW_WIDTH + light.radius || point.y > VIEW_HEIGHT + light.radius) continue;
    const gradient = lightContext.createRadialGradient(point.x, point.y, 8, point.x, point.y, light.radius);
    gradient.addColorStop(0, 'rgba(0,0,0,0.96)');
    gradient.addColorStop(0.45, 'rgba(0,0,0,0.72)');
    gradient.addColorStop(1, 'rgba(0,0,0,0)');
    lightContext.fillStyle = gradient;
    lightContext.beginPath();
    lightContext.arc(point.x, point.y, light.radius, 0, Math.PI * 2);
    lightContext.fill();
  }
  lightContext.globalCompositeOperation = 'source-over';
  context.drawImage(lightCanvas, 0, 0);
}

function currentContextHint() {
  const nearbySalvage = world.salvage.find((salvage) => salvage.amount > 0 && distance(world.player, salvage) < 85);
  if (nearbySalvage) return `E  Recover ${nearbySalvage.amount} Scrap`;
  if (distance(world.player, world.heartforge) < 135) {
    if (!world.heartforge.restored) {
      return world.scrap >= HEARTFORGE_REPAIR_COST
        ? 'E  Restore the Heartforge'
        : `Need ${HEARTFORGE_REPAIR_COST - world.scrap} more Scrap`;
    }
    if (!world.progress.evolution) return 'E  Choose an evolution';
    if (world.progress.expeditionStage === 'returned' && !world.progress.cognitionInstalled) return 'E  Install the Cognition Core';
  }
  if (canAuthorizeExpedition(world)) return 'X  Authorize North Ruins expedition';
  return 'AUTO-FIRE ACTIVE';
}
