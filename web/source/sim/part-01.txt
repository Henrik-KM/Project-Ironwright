function clamp(value, min, max) {
  return Math.max(min, Math.min(max, value));
}

function length(x, y) {
  return Math.hypot(x, y);
}

function distance(a, b) {
  return Math.hypot(a.x - b.x, a.y - b.y);
}

function normalized(x, y) {
  const magnitude = Math.hypot(x, y);
  if (magnitude <= 0.00001) return { x: 0, y: 0 };
  return { x: x / magnitude, y: y / magnitude };
}

function random(world) {
  let x = world.rngState | 0;
  x ^= x << 13;
  x ^= x >>> 17;
  x ^= x << 5;
  world.rngState = x >>> 0;
  return (world.rngState & 0xffffffff) / 0x100000000 + 0.5;
}

function randomRange(world, min, max) {
  return min + (max - min) * random(world);
}

function nextId(world, prefix) {
  const id = `${prefix}-${world.nextEntityId}`;
  world.nextEntityId += 1;
  return id;
}

function pushEvent(world, text, kind = 'info') {
  world.events.unshift({ id: nextId(world, 'event'), time: world.time, text, kind });
  if (world.events.length > MAX_EVENTS) world.events.length = MAX_EVENTS;
}

function circleIntersectsRect(x, y, radius, rect) {
  const nearestX = clamp(x, rect.x, rect.x + rect.w);
  const nearestY = clamp(y, rect.y, rect.y + rect.h);
  return Math.hypot(x - nearestX, y - nearestY) < radius;
}

function canOccupy(world, x, y, radius) {
  if (x < radius || y < radius || x > WORLD_WIDTH - radius || y > WORLD_HEIGHT - radius) return false;
  for (const obstacle of world.obstacles) {
    if (circleIntersectsRect(x, y, radius, obstacle)) return false;
  }
  return true;
}

function moveWithCollisions(world, entity, dx, dy, radius = 16) {
  const nextX = clamp(entity.x + dx, radius, WORLD_WIDTH - radius);
  if (canOccupy(world, nextX, entity.y, radius)) entity.x = nextX;
  const nextY = clamp(entity.y + dy, radius, WORLD_HEIGHT - radius);
  if (canOccupy(world, entity.x, nextY, radius)) entity.y = nextY;
}

function moveIgnoringBuildings(entity, dx, dy, radius = 16) {
  entity.x = clamp(entity.x + dx, radius, WORLD_WIDTH - radius);
  entity.y = clamp(entity.y + dy, radius, WORLD_HEIGHT - radius);
}
