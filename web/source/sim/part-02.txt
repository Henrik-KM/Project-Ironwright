function findEnemyById(world, id) {
  return world.enemies.find((enemy) => enemy.id === id && !enemy.dead) ?? null;
}

function findRobotById(world, id) {
  return world.robots.find((robot) => robot.id === id && !robot.disabled) ?? null;
}

function nearestEnemy(world, origin, range = Infinity) {
  let best = null;
  let bestDistance = range;
  for (const enemy of world.enemies) {
    if (enemy.dead || enemy.hp <= 0) continue;
    const currentDistance = distance(origin, enemy);
    if (currentDistance < bestDistance) {
      best = enemy;
      bestDistance = currentDistance;
    }
  }
  return best;
}

function nearestSalvage(world, origin, range = Infinity, requireUnreserved = false) {
  let best = null;
  let bestDistance = range;
  for (const salvage of world.salvage) {
    if (salvage.amount <= 0) continue;
    if (requireUnreserved && salvage.reservedBy) continue;
    const currentDistance = distance(origin, salvage);
    if (currentDistance < bestDistance) {
      best = salvage;
      bestDistance = currentDistance;
    }
  }
  return best;
}

function spawnProjectile(world, ownerKind, owner, target, damage, speed, color) {
  if (world.projectiles.length >= MAX_PROJECTILES) world.projectiles.shift();
  const direction = normalized(target.x - owner.x, target.y - owner.y);
  world.projectiles.push({
    id: nextId(world, 'shot'),
    ownerKind,
    ownerId: owner.id,
    targetId: target.id,
    x: owner.x + direction.x * (owner.radius + 5),
    y: owner.y + direction.y * (owner.radius + 5),
    vx: direction.x * speed,
    vy: direction.y * speed,
    damage,
    radius: ownerKind === 'player' ? 5 : 4,
    ttl: 1.2,
    color,
  });
  owner.heading = Math.atan2(direction.y, direction.x);
}

function updatePlayer(world, input, dt) {
  const player = world.player;
  player.fireCooldown = Math.max(0, player.fireCooldown - dt);
  player.dashCooldown = Math.max(0, player.dashCooldown - dt);
  player.dashTime = Math.max(0, player.dashTime - dt);
  player.invulnerable = Math.max(0, player.invulnerable - dt);

  if (input.dash && player.dashCooldown <= 0) {
    player.dashCooldown = 1.35;
    player.dashTime = 0.17;
    player.invulnerable = 0.23;
  }

  const movement = normalized(Number(input.moveX) || 0, Number(input.moveY) || 0);
  const speedMultiplier = player.dashTime > 0 ? 3.25 : 1;
  const dx = movement.x * player.speed * speedMultiplier * dt;
  const dy = movement.y * player.speed * speedMultiplier * dt;
  if (Math.abs(dx) + Math.abs(dy) > 0) {
    moveWithCollisions(world, player, dx, dy, player.radius);
    if (!player.targetId) player.heading = Math.atan2(movement.y, movement.x);
  }

  const target = nearestEnemy(world, player, player.fireRange);
  player.targetId = target?.id ?? null;
  if (target) {
    player.heading = Math.atan2(target.y - player.y, target.x - player.x);
    if (player.fireCooldown <= 0) {
      spawnProjectile(world, 'player', player, target, player.damage, 920, '#7fe8ff');
      player.fireCooldown = player.fireInterval;
    }
  } else if (Number.isFinite(input.aimX) && Number.isFinite(input.aimY)) {
    const aim = normalized(input.aimX - player.x, input.aimY - player.y);
    if (Math.abs(aim.x) + Math.abs(aim.y) > 0) player.heading = Math.atan2(aim.y, aim.x);
  }
}

function chooseLocalRobotThreat(world, robot) {
  const local = nearestEnemy(world, robot, robot.range);
  if (local) return local;
  const baseThreat = nearestEnemy(world, world.heartforge, 520);
  if (baseThreat && distance(robot, world.heartforge) < 750) return baseThreat;
  return null;
}

function moveToward(world, entity, target, speed, dt, collision = true, radius = 14) {
  const direction = normalized(target.x - entity.x, target.y - entity.y);
  entity.heading = Math.atan2(direction.y, direction.x);
  if (collision) moveWithCollisions(world, entity, direction.x * speed * dt, direction.y * speed * dt, radius);
  else moveIgnoringBuildings(entity, direction.x * speed * dt, direction.y * speed * dt, radius);
  return distance(entity, target);
}
